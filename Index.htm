    function switchTab(tabId) {

      document.querySelectorAll('.tab-content').forEach(t => t.classList.remove('active'));

      document.querySelectorAll('.nav button').forEach(b => b.classList.remove('active'));

      document.getElementById(tabId).classList.add('active');

     

      if(tabId === 'entreno') document.getElementById('tab1-btn').classList.add('active');

      if(tabId === 'progreso') {

        document.getElementById('tab2-btn').classList.add('active');

        fetchHistory();

      }

    }

 

    function loadRoutine() {

      const selectedDay = document.getElementById("day-select").value;

      const container = document.getElementById("routine-container");

      container.innerHTML = "";

 

      const currentRoutine = routines[selectedDay];

     

      if (currentRoutine.length === 0) {

        container.innerHTML = `<div class="card" style="text-align:center; color:var(--text-dim);">Pure Running Session. Focus on distance and pace below.</div>`;

        return;

      }

 

      let currentSuperset = "";

      let currentCard = null;

 

      currentRoutine.forEach((ex, idx) => {

        if (ex.superset !== currentSuperset) {

          currentSuperset = ex.superset;

          currentCard = document.createElement("div");

          currentCard.className = "card";

          currentCard.innerHTML = `<div class="superset-title">${currentSuperset}</div>`;

          container.appendChild(currentCard);

        }

 

        const exDiv = document.createElement("div");

        exDiv.className = "exercise-item";

        exDiv.innerHTML = `

          <div class="exercise-header">

            <span class="exercise-name">${ex.name}</span>

          </div>

          <div class="note-box">Target: ${ex.defaultReps}</div>

          ${[1, 2, 3].map(set => `

            <div class="set-row">

              <label>Set ${set}</label>

              <input type="number" id="peso_${idx}_${set}" placeholder="Weight">

              <input type="number" id="reps_${idx}_${set}" placeholder="Reps">

            </div>

          `).join('')}

        `;

        currentCard.appendChild(exDiv);

      });

    }

 

    async function saveWorkout() {

      const selectedDay = document.getElementById("day-select").value;

      const currentRoutine = routines[selectedDay];

      let ejercicios = [];

 

      currentRoutine.forEach((ex, idx) => {

        [1, 2, 3].forEach(set => {

          const peso = document.getElementById(`peso_${idx}_${set}`).value;

          const reps = document.getElementById(`reps_${idx}_${set}`).value;

          if (peso || reps) {

            ejercicios.push({

              nombre: ex.name,

              serie: set,

              peso: peso || 0,

              reps: reps || 0

            });

          }

        });

      });

 

      const cardioDist = document.getElementById("cardio-dist").value;

      const cardioPace = document.getElementById("cardio-pace").value;

      const cardioNotes = document.getElementById("cardio-notes").value;

 

      if (ejercicios.length === 0 && !cardioDist) {

        alert("Please enter at least one weight/reps value or cardio data.");

        return;

      }

 

      const payload = {

        dia: selectedDay,

        ejercicios: ejercicios,

        cardio: { distancia: cardioDist, pace: cardioPace, notes: cardioNotes }

      };

 

      try {

        await fetch(SCRIPT_URL, {

          method: "POST",

          mode: "no-cors",

          headers: { "Content-Type": "application/json" },

          body: JSON.stringify(payload)

        });

        alert("Workout successfully saved to Google Sheets.");

        loadRoutine();

        document.getElementById("cardio-dist").value = "";

        document.getElementById("cardio-pace").value = "";

        document.getElementById("cardio-notes").value = "";

      } catch (e) {

        alert("Error saving workout: " + e.toString());

      }

    }

 

    async function fetchHistory() {

      try {

        const res = await fetch(SCRIPT_URL);

        const data = await res.json();

        historyData = data.slice(1); // Skip headers

 

        // Index 7 corresponds to Exercise Name in the new columns layout

        const uniqueExercises = [...new Set(historyData.map(r => r[7]))].filter(Boolean);

        const select = document.getElementById("chart-exercise-select");

        select.innerHTML = uniqueExercises.map(ex => `<option value="${ex}">${ex}</option>`).join('');

       

        updateChart();

      } catch(e) {

        console.error("Error fetching history", e);

      }

    }

 

    function updateChart() {

      const selectedEx = document.getElementById("chart-exercise-select").value;

      // Filter by Exercise Name (index 7)

      const filtered = historyData.filter(r => r[7] === selectedEx);

 

      // Index 1 is Date (YYYY-MM-DD), Index 9 is Weight, Index 11 is Cardio Distance

      const labels = filtered.map(r => r[1]);

      const values = filtered.map(r => parseFloat(r[9]) || parseFloat(r[11]) || 0);

 

      const ctx = document.getElementById('progressChart').getContext('2d');

      if (chartInstance) chartInstance.destroy();

 

      chartInstance = new Chart(ctx, {

        type: 'line',

        data: {

          labels: labels,

          datasets: [{

            label: selectedEx.includes('Cardio') ? 'Distance (km)' : 'Weight Used',

            data: values,

            borderColor: '#f97316',

            backgroundColor: 'rgba(249, 115, 22, 0.15)',

            borderWidth: 2,

            fill: true,

            tension: 0.2

          }]

        },

        options: {

          responsive: true,

          scales: {

            y: { beginAtZero: true, grid: { color: '#27272a' }, ticks: { color: '#9ca3af' } },

            x: { grid: { color: '#27272a' }, ticks: { color: '#9ca3af' } }

          },

          plugins: {

            legend: { labels: { color: '#f3f4f6' } }

          }

        }

      });

    }
